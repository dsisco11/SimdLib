cmake_minimum_required(VERSION 3.31)
project(CodegenCompositionHarness LANGUAGES NONE)
enable_testing()
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
include("${root}/cmake/development/CodegenTests.cmake")
file(COPY_FILE "${PROBE_OBJECT}" "${CMAKE_CURRENT_BINARY_DIR}/probe.o")
set(object "${CMAKE_CURRENT_BINARY_DIR}/probe.o")
add_library(SharedFixture OBJECT IMPORTED)
set_target_properties(SharedFixture PROPERTIES IMPORTED_OBJECTS "${object}")
set(artifact_root "${CMAKE_CURRENT_BINARY_DIR}/artifacts")
set(manifest "${CMAKE_CURRENT_BINARY_DIR}/inputs.cmake")
set(receipt "${CMAKE_CURRENT_BINARY_DIR}/build.sha256")
set(configuration "${CMAKE_CURRENT_BINARY_DIR}/configuration.txt")
file(WRITE "${configuration}" "compiler=prebuilt extraction harness\nconfiguration=composition\n")
file(WRITE "${manifest}" "set(INPUTS [==[${configuration};${SIMDLIB_FILECHECK};${SIMDLIB_LLVM_OBJDUMP};${SIMDLIB_LLVM_READOBJ}]==])\n")
set_target_properties(SharedFixture PROPERTIES
    SIMDLIB_CODEGEN_ROOT "${artifact_root}" SIMDLIB_CODEGEN_CONFIGURATION probe
    SIMDLIB_CODEGEN_GEOMETRY identity SIMDLIB_CODEGEN_INPUT_MANIFEST "${manifest}"
    SIMDLIB_CODEGEN_BUILD_RECEIPT "${receipt}" SIMDLIB_CODEGEN_CONFIGURATION_FILE "${configuration}")
set(primary "${CMAKE_CURRENT_BINARY_DIR}/primary.cmake")
set(supplement "${CMAKE_CURRENT_BINARY_DIR}/supplement.cmake")
set(return_rule "${root}/tests/codegen/contracts/return.check")
set(wrong_rule "${CMAKE_CURRENT_BINARY_DIR}/wrong.check")
file(WRITE "${wrong_rule}" "CHECK: deliberate_missing_instruction\n")
foreach(kind IN ITEMS primary supplement)
    if(CASE STREQUAL "failed-${kind}")
        set(rule "${wrong_rule}")
    else()
        set(rule "${return_rule}")
    endif()
    file(WRITE "${${kind}}" "set(RULES [==[${rule}]==])\n")
endforeach()
simdlib_add_codegen_case(ID composition TARGET SharedFixture DRIVER clang
    PRIMARY "${primary}" SYMBOLS extraction_identity
    SUPPLEMENTS "fact|clang|probe|${supplement}")
simdlib_finalize_codegen_cases(Codegen.composition.identity.probe.primary
    Codegen.composition.identity.probe.supplemental.fact)
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT_MANIFEST=${manifest}"
    "-DOBJECT_FILE=${object}" "-DBUILD_RECEIPT=${receipt}"
    -P "${root}/cmake/codegen/RecordBuild.cmake" COMMAND_ERROR_IS_FATAL ANY)
if(CASE STREQUAL "stale-configuration")
    file(APPEND "${configuration}" "changed=true\n")
elseif(CASE STREQUAL "stale-object")
    file(APPEND "${object}" "changed")
elseif(CASE STREQUAL "missing-object")
    file(REMOVE "${object}")
elseif(CASE STREQUAL "missing-receipt")
    file(REMOVE "${receipt}")
endif()
