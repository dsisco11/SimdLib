cmake_minimum_required(VERSION 3.31)
project(CodegenRegistrationHarness LANGUAGES NONE)
enable_testing()
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
include("${root}/cmake/development/CodegenTests.cmake")
add_library(SharedFixture OBJECT IMPORTED)
set_target_properties(SharedFixture PROPERTIES IMPORTED_OBJECTS "${CMAKE_CURRENT_BINARY_DIR}/unused.o")
foreach(key IN ITEMS ROOT INPUT_MANIFEST BUILD_RECEIPT CONFIGURATION_FILE)
    set_property(TARGET SharedFixture PROPERTY SIMDLIB_CODEGEN_${key} "${CMAKE_CURRENT_BINARY_DIR}/${key}")
endforeach()
set_property(TARGET SharedFixture PROPERTY SIMDLIB_CODEGEN_CONFIGURATION probe)
set_property(TARGET SharedFixture PROPERTY SIMDLIB_CODEGEN_GEOMETRY geometry)
set(primary "${CMAKE_CURRENT_BINARY_DIR}/primary.cmake")
file(WRITE "${primary}" "set(RULES unused.check)\n")
set(selectors "clang(-cl)?")
if(CASE STREQUAL "wrong-applicability")
    set(selectors msvc)
endif()
set(supplements "fact|${selectors}|probe|${primary}")
if(CASE STREQUAL "missing-supplement")
    set(supplements "")
endif()
if(NOT CASE STREQUAL "missing-primary")
    simdlib_add_codegen_case(ID probe TARGET SharedFixture DRIVER clang-cl
        PRIMARY "${primary}" SYMBOLS probe_caller probe_callee SUPPLEMENTS ${supplements})
endif()
set(expected Codegen.probe.geometry.probe.primary Codegen.probe.geometry.probe.supplemental.fact)
simdlib_finalize_codegen_cases(${expected})
get_property(primary_fixture TEST Codegen.probe.geometry.probe.primary PROPERTY FIXTURES_SETUP)
get_property(supplement_fixture TEST Codegen.probe.geometry.probe.supplemental.fact PROPERTY FIXTURES_SETUP)
get_property(required_fixture TEST Codegen.probe.geometry.probe.qualified PROPERTY FIXTURES_REQUIRED)
if(NOT primary_fixture STREQUAL required_fixture OR NOT supplement_fixture STREQUAL required_fixture)
    message(FATAL_ERROR "Qualification must require both primary and supplemental tests")
endif()
